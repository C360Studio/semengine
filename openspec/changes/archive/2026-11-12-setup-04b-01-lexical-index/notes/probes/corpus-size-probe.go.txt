package embedding

import (
	"bytes"
	"compress/gzip"
	"encoding/json"
	"math/rand"
	"regexp"
	"testing"
)

var issueRef = regexp.MustCompile(`#[0-9]{2,}`)

func TestCorpusSize(t *testing.T) {
	secs := loadSections(t, "../..")
	rng := rand.New(rand.NewSource(1))
	rng.Shuffle(len(secs), func(i, j int) { secs[i], secs[j] = secs[j], secs[i] })
	secs = secs[:2000]
	var raw bytes.Buffer
	enc := json.NewEncoder(&raw)
	for _, s := range secs {
		_ = enc.Encode(map[string]string{"path": s.path, "heading": s.heading, "body": s.body})
	}
	var gz bytes.Buffer
	w, _ := gzip.NewWriterLevel(&gz, gzip.BestCompression)
	_, _ = w.Write(raw.Bytes())
	_ = w.Close()
	hist := 0
	for _, s := range secs {
		if issueRef.MatchString(s.body) {
			hist++
		}
	}
	t.Logf("2000 sections: %d bytes as JSON lines, %d bytes gzipped; %d sections cite an issue or pull request number", raw.Len(), gz.Len(), hist)
}
